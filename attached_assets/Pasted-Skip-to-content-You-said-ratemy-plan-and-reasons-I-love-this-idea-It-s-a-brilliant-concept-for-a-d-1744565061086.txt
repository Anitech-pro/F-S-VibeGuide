// src/components/sections/FrontendSection.tsx
'use client';

import { useDispatch, useSelector } from 'react-redux';
import { selectTechOption } from '@/store/stackSelectionSlice';
import { RootState } from '@/store/store';

const frontendOptions = [
  { id: 'react', label: 'React' },
  { id: 'vue', label: 'Vue.js' },
  { id: 'svelte', label: 'Svelte' },
];

export function FrontendSection() {
  const dispatch = useDispatch();
  const selectedFrontend = useSelector((state: RootState) => state.stackSelection.selected.frontend);

  return (

      <h2 className="text-xl font-semibold mb-4">Choose Frontend Framework</h2>
      <div className="grid grid-cols-2 gap-4">
        {frontendOptions.map((option) => (

            key={option.id}
            className={`p-4 rounded-xl border-2 ${
              selectedFrontend === option.id
                ? 'border-blue-500 bg-blue-100'
                : 'border-gray-300 hover:border-blue-300'
            }`}
            onClick={() => dispatch(selectTechOption({ section: 'frontend', option: option.id }))}

            {option.label}

// src/store/stackSelectionSlice.ts
import { createSlice, PayloadAction } from '@reduxjs/toolkit';

interface StackState {
  stackType: 'web' | 'crossPlatform' | 'nativeIOS' | 'nativeAndroid' | null;
  selected: Record<string, string>; // key = section name, value = selected tech id
}

const initialState: StackState = {
  stackType: null,
  selected: {},
};

const stackSelectionSlice = createSlice({
  name: 'stackSelection',
  initialState,
  reducers: {
    setStackType(state, action: PayloadAction<StackState['stackType']>) {
      state.stackType = action.payload;
      state.selected = {}; // Reset selected when type changes
    },
    selectTechOption(state, action: PayloadAction<{ section: string; option: string }>) {
      const { section, option } = action.payload;
      state.selected[section] = option;
    },
    clearSelections(state) {
      state.selected = {};
    },
  },
});

export const { setStackType, selectTechOption, clearSelections } = stackSelectionSlice.actions;
export default stackSelectionSlice.reducer;
